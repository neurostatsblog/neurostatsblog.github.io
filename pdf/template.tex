% Title block — arxiv-style with colored rule
%======================================================================
\makeatletter

\newcommand{\nsbtitleblock}{%
  \begingroup
  \centering
  {\LARGE\bfseries \@title \par}%
  \vspace{0.5em}%
  {\color{nsbprimary}\rule{\linewidth}{0.6pt}}\par
  $if(subtitle)$
  \vspace{0.6em}%
  {\large\itshape $subtitle$\par}%
  $endif$
  \vspace{1.4em}%
  $if(authors)$
  {\normalsize
  $for(authors)$%
  \begin{minipage}[t]{0.95\linewidth}\centering
    {$authors.name$}\\[2pt]
    $if(authors.affiliation)$\emph{$authors.affiliation$}\\[2pt]$endif$
    $if(authors.email)$\texttt{\small $authors.email$}$endif$
  \end{minipage}%
  $sep$\hspace{2em}%
  $endfor$
  \par}
  $endif$
  $if(date)$
  \vspace{1.0em}%
  {\small\itshape $date$\par}%
  $endif$
  \vspace{1.8em}%
  \endgroup
}

\makeatother
